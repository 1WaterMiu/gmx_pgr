cmake_minimum_required(VERSION 3.18)
project(gmx_pgr VERSION 0.1.0 LANGUAGES CXX)
set(GROMACS_SOURCE "" CACHE PATH "Source tree matching the installed GROMACS version")
set(GROMACS_PREFIX "" CACHE PATH "Mixed-precision GROMACS installation prefix")
if(NOT GROMACS_PREFIX)
    find_program(GMX_EXECUTABLE NAMES gmx)
    if(NOT GMX_EXECUTABLE)
        message(FATAL_ERROR "Set -DGROMACS_PREFIX=/path/to/gromacs-install")
    endif()
    get_filename_component(_gmx_bin "${GMX_EXECUTABLE}" DIRECTORY)
    get_filename_component(GROMACS_PREFIX "${_gmx_bin}" DIRECTORY)
endif()
if(GROMACS_PREFIX STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Point GROMACS_PREFIX at the original GROMACS installation, not this extension.")
endif()
find_library(GROMACS_LIBRARY NAMES gromacs PATHS "${GROMACS_PREFIX}/lib" "${GROMACS_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${GROMACS_PREFIX}/bin/gmx")
    message(FATAL_ERROR "Expected the original executable at GROMACS_PREFIX/bin/gmx")
endif()
execute_process(COMMAND "${GROMACS_PREFIX}/bin/gmx" --version OUTPUT_VARIABLE _gmx_version ERROR_VARIABLE _gmx_version_err RESULT_VARIABLE _gmx_rc)
string(APPEND _gmx_version "${_gmx_version_err}")
if(NOT _gmx_rc EQUAL 0 OR NOT _gmx_version MATCHES "Precision:[ \t]+mixed")
    message(FATAL_ERROR "This build requires a runnable, mixed-precision GROMACS installation.")
endif()
if(NOT _gmx_version MATCHES "GROMACS version:[ \t]+2025\\.3")
    message(WARNING "Only GROMACS 2025.3 is tested. Matching headers and library are required; API compatibility is not guaranteed.")
endif()
if(GROMACS_SOURCE)
    set(_gmx_includes "${GROMACS_SOURCE}/api/legacy/include" "${GROMACS_SOURCE}/src" "${GROMACS_SOURCE}/src/external")
    if(NOT EXISTS "${GROMACS_SOURCE}/api/legacy/include/gromacs/trajectoryanalysis/analysismodule.h")
        message(FATAL_ERROR "GROMACS_SOURCE does not contain the required analysis headers")
    endif()
elseif(EXISTS "${GROMACS_PREFIX}/include/gromacs/trajectoryanalysis/analysismodule.h")
    set(_gmx_includes "${GROMACS_PREFIX}/include")
else()
    message(FATAL_ERROR "Analysis headers are missing; set -DGROMACS_SOURCE=/path/to/matching-gromacs-source")
endif()
add_executable(gmx_pgr pgr.cpp main.cpp)
target_compile_features(gmx_pgr PRIVATE cxx_std_17)
target_compile_options(gmx_pgr PRIVATE -O3 -Wall -Wextra)
target_include_directories(gmx_pgr PRIVATE ${_gmx_includes} "${CMAKE_CURRENT_SOURCE_DIR}/compat")
target_link_libraries(gmx_pgr PRIVATE "${GROMACS_LIBRARY}")
get_filename_component(_gmx_libdir "${GROMACS_LIBRARY}" DIRECTORY)
set_target_properties(gmx_pgr PROPERTIES BUILD_RPATH "${_gmx_libdir}" OUTPUT_NAME gmx RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
target_compile_definitions(gmx_pgr PRIVATE GMX_ORIGINAL="${GROMACS_PREFIX}/bin/gmx")
